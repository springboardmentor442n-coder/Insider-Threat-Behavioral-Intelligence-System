# --- Authentication ----------------------------------------------------------
# NOTE: we use `bcrypt` DIRECTLY rather than passlib.
# passlib 1.7.4 (the usual recommendation, and what most tutorials show) is
# unmaintained and CRASHES against bcrypt >= 4.1 with a confusing
# "password cannot be longer than 72 bytes" error caused by its own broken
# version detection. This was verified empirically before any auth code was
# written. Using bcrypt directly is one fewer dependency and one fewer
# thing to break.
bcrypt==4.2.1
python-jose[cryptography]==3.3.0   # JWT encode/decode
